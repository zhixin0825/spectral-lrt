\section{Proofs of spectral preservation and statistical results}
\label{app:statistical-proofs}

\subsection{Exact preservation and the population contrast space}

\begin{proof}[Proof of Proposition~\ref{prop:exact-preservation}]
Since $AU=U\Lambda$, multiplication by $U^\top$ gives
$A\Pi=AUU^\top=U\Lambda U^\top=\widetilde A$.  Therefore
$\widetilde A_iw-A_iw=A_i(\Pi-\mathrm{Id})w$.  If $w\in\operatorname{col}(U)$,
then $\Pi w=w$ and the difference vanishes.

Let $N=Z^\top Z=\operatorname{diag}(n_1,\ldots,n_K)$, which is
nonsingular since all communities are nonempty.  The columns of
$ZN^{-1/2}$ are orthonormal, and
\[
 ZPZ^\top=(ZN^{-1/2})(N^{1/2}PN^{1/2})(ZN^{-1/2})^\top.
\]
Nonsingularity of $P$ makes the central matrix nonsingular.  Hence
$ZPZ^\top$ has rank $K$ and column space $\operatorname{col}(Z)$.
For each contrast, define
$\xi_c^{ab}=\log\{P_{ac}(1-P_{bc})/[P_{bc}(1-P_{ac})]\}$.
Then $w^{ab}=Z\xi^{ab}$, which proves membership in that space.
Finally, subtracting $\operatorname{diag}(P_{z_i z_i})$ from $ZPZ^\top$
gives the actual no-self-loop mean, so no rank assertion about that
corrected matrix is needed.
\end{proof}

Full rank cannot be replaced by distinct block rows without a further
contrast-span condition.  For example, let $P=ss^\top$ for a positive,
nonconstant vector $s$.  Its signal space is spanned by $Zs$, whereas
the sparse log-probability contrast between communities $a,b$ is
$\log(s_a/s_b)\mathbf1$.  When $s_a\ne s_b$, this constant vector does
not belong to $\operatorname{span}(Zs)$.  A rank-one signal projection
does not preserve that contrast by the preceding identity.  This example
only concerns projection onto the population rank-one space; it is not
an impossibility claim for every statistic computed from one empirical
eigenvector.


\subsection{The sparse likelihood correction}

\begin{proof}[Proof of Lemma~\ref{lem:sparse-expansion}]
Write $p_{aj}=P_{a,z_j}$ and $p_{bj}=P_{b,z_j}$ for $j\ne i$.
The exact binary Bernoulli log-likelihood ratio is
\[
 L_i^{ab}=\sum_{j\ne i}A_{ij}\log\frac{p_{aj}}{p_{bj}}
 +\sum_{j\ne i}A_{ij}\log\frac{1-p_{bj}}{1-p_{aj}}
 +\sum_{j\ne i}\log\frac{1-p_{aj}}{1-p_{bj}}.
\]
For $0\le x\le1/2$, the power series for $\log(1-x)$ implies
$|\log(1-x)+x|\le Cx^2$. Since $p_{aj},p_{bj}\le P_{\max}\to0$,
\[
 \left|\log\frac{1-p_{bj}}{1-p_{aj}}\right|\le C P_{\max},
 \qquad
 \left|\log\frac{1-p_{aj}}{1-p_{bj}}+(p_{aj}-p_{bj})\right|
 \le C P_{\max}^2.
\]
The first term is unchanged in the sparse score, and the final term is
replaced by its first-order nonedge offset. Summing proves
\eqref{eq:sparse-expansion-error}.

Under any true community, $\deg(i)$ is a sum of independent Bernoulli
variables with mean at most $nP_{\max}$. For $D>1$, exponential Markov
inequality with $t=\log D$ gives
\[
 \Pr\{\deg(i)>D nP_{\max}\}
 \le\exp\{-nP_{\max}(D\log D-D+1)\}.
\]
The bracket tends to infinity with $D$. Thus, for every fixed $H>0$,
the degree bound holds uniformly in $i$ outside probability
$e^{-H nP_{\max}}$; a union bound has the same form after increasing
$D$, because $nP_{\max}\ge c\log n$.
On that event the likelihood correction is
$O_H(nP_{\max}^2)=o(nP_{\max})$.
This error need not tend to zero in absolute value: its relevant scale
is the information $I_n\asymp nP_{\max}$ established below.
\end{proof}

\begin{remark}[Scope of the sparse working score]
The density lower bound permits expected degrees larger than $\log n$.
The assumption $P_{\max}\to0$ is used separately to make the first-order
Bernoulli correction negligible relative to $I_n$. No result here
identifies this sparse score with the exact Bernoulli likelihood when
the block probabilities stay bounded away from zero.
\end{remark}


\subsection{Chernoff information and the oracle exponent}

\begin{proof}[Proof of Lemma~\ref{lem:oracle-exponent}]
Let $f_a,f_b$ denote the product Bernoulli mass functions of the
incident-edge row under the two hypotheses. Define
\begin{align*}
 D_\alpha(p_{a,-i}\Vert p_{b,-i})
 &=-\log\sum_x f_a(x)^{1-\alpha}f_b(x)^\alpha\\
 &=-\sum_{j\ne i}\log\big\{p_{aj}^{1-\alpha}p_{bj}^\alpha
                    +(1-p_{aj})^{1-\alpha}(1-p_{bj})^\alpha\big\},\\
 D^*_{i,ab}&=\max_{0\le\alpha\le1}D_\alpha(p_{a,-i}\Vert p_{b,-i}).
\end{align*}
Uniformly for $\alpha\in[0,1]$ and $u,v\in[P_{\min},P_{\max}]$,
\[
 u^{1-\alpha}v^\alpha+(1-u)^{1-\alpha}(1-v)^\alpha
 =1-\{(1-\alpha)u+\alpha v-u^{1-\alpha}v^\alpha\}
   +O(P_{\max}^2).
\]
For example, the nonedge product expansion follows by writing it as
$\exp\{(1-\alpha)\log(1-u)+\alpha\log(1-v)\}$.
Expanding $-\log(1-x)$ and summing gives
\begin{equation}
 D^*_{i,ab}
 =n\max_{0\le\alpha\le1}\sum_c\pi_{c,n}
  \{(1-\alpha)P_{ac}+\alpha P_{bc}-P_{ac}^{1-\alpha}P_{bc}^\alpha\}
  +O(nP_{\max}^2+P_{\max}),\label{eq:chernoff-expansion}
\end{equation}
where $\pi_{c,n}=n_c/n$. The $O(P_{\max})$ term accounts for the
excluded node. A fixed limiting normalized matrix is unnecessary.
Indeed, $\sigma_{\min}(N^{1/2}PN^{1/2})\ge c nP_{\max}$ implies
$\sigma_{\min}(P)\ge cP_{\max}$, since $\|N^{1/2}\|_{\mathrm{op}}^2\le n$.
Consequently, for every $a\ne b$,
\[
 \sum_c\pi_{c,n}(\sqrt{P_{ac}}-\sqrt{P_{bc}})^2
 \ge\frac{\pi_{\min}}{4P_{\max}}\|P_{a*}-P_{b*}\|_2^2
 \ge c'P_{\max}.
\]
Here $P_{a*}$ denotes the $K$-coordinate block row in this display.
Taking $\alpha=1/2$ in \eqref{eq:chernoff-expansion} proves a uniform
lower bound $D^*_{i,ab}\ge c' nP_{\max}$ for large $n$.
The upper bound $D^*_{i,ab}\le C nP_{\max}$ follows from the same
expansion. Thus $D^*_{i,ab}\asymp nP_{\max}$ and
$I_n\asymp nP_{\max}$, uniformly over all pairs and excluded nodes.

The equal-prior Bayes risk is
$R_{i,ab}^{\mathrm{or}}=\frac12\sum_x\min\{f_a(x),f_b(x)\}$.
Since $\min\{u,v\}\le u^{1-\alpha}v^\alpha$,
\begin{equation}
 R_{i,ab}^{\mathrm{or}}\le\tfrac12 e^{-D^*_{i,ab}}.
 \label{eq:oracle-upper}
\end{equation}
A matching leading lower bound follows by exponential tilting. Let
$\alpha_*$ maximize the Chernoff information and set
\[
 f_*(x)=e^{D^*_{i,ab}}f_a(x)^{1-\alpha_*}f_b(x)^{\alpha_*},
 \qquad L(x)=\log\frac{f_a(x)}{f_b(x)}.
\]
The two positive product distributions differ and the endpoint
derivatives have opposite signs. Hence $\alpha_*\in(0,1)$, and
differentiating the normalization gives $\mathbb E_*L=0$.
The tilted incident edges remain independent, with probabilities
\[
 p_{*,j}=
 \frac{p_{aj}^{1-\alpha_*}p_{bj}^{\alpha_*}}
 {p_{aj}^{1-\alpha_*}p_{bj}^{\alpha_*}
 +(1-p_{aj})^{1-\alpha_*}(1-p_{bj})^{\alpha_*}}
 \le C P_{\max}.
\]
The edge coefficient in $L$ is uniformly bounded because
$P_{\min}\ge cP_{\max}$ and $P_{\max}\to0$.
Thus $\operatorname{Var}_*(L)\le C nP_{\max}$.
Take $h_n=(nP_{\max})^{2/3}$. Chebyshev's inequality gives
$\Pr_*(|L|\le h_n)=1-o(1)$, and the change of measure yields
\begin{align*}
 R_{i,ab}^{\mathrm{or}}
 &=\tfrac12 e^{-D^*_{i,ab}}
   \mathbb E_*\min\{e^{\alpha_*L},e^{-(1-\alpha_*)L}\}\\
 &\ge\tfrac12 e^{-D^*_{i,ab}-h_n}\Pr_*(|L|\le h_n).
\end{align*}
Together with \eqref{eq:oracle-upper}, this proves
$\log R_{i,ab}^{\mathrm{or}}=-D^*_{i,ab}+o(nP_{\max})$ and hence
\eqref{eq:oracle-risk}. Fixed prior probabilities bounded away from
zero and one change this risk by at most fixed factors, preserving
the leading exponent.
\end{proof}

\subsection{Transfer of the leading exponent}

\begin{proof}[Proof of Theorem~\ref{thm:exponent-transfer}]
Suppose the true label of node $i$ is $a$.  If $G_i$ occurs and the
classifier prefers $b\ne a$, then
$\widehat\ell_i(a)-\widehat\ell_i(b)\le0$, and consequently
$L_i^{ab}\le\tau_n$.  For every $\alpha\in[0,1]$,
\[
 \Pr_a\{L_i^{ab}\le\tau_n\}
 \le e^{\alpha\tau_n}\mathbb E_a e^{-\alpha L_i^{ab}}
 =\exp\{\alpha\tau_n-D_\alpha(p_{a,-i}\Vert p_{b,-i})\}.
\]
Selecting a Chernoff maximizer and using $\alpha\le1$ gives
\[
 \Pr\{\widehat z_i\ne z_i\}
 \le\delta_n+\sum_{b\ne a}\exp\{\tau_n-D^*_{i,ab}\}
 \le\delta_n+(K-1)\exp\{-I_n+o(I_n)\}.
\]
Because $\tau_n=o(I_n)$,
\eqref{eq:transfer-tail} makes the first term no larger in leading
exponential order.  Averaging over nodes and using that the
permutation-invariant error is no larger than the error under any one
common alignment proves \eqref{eq:conditional-rate}.

The expected number of errors under that alignment is at most
$n\exp\{-(1+o(1))I_n\}$. If
$\liminf_{n\to\infty}I_n/\log n>1$, Markov's inequality gives vanishing
probability of even one error, which proves the exact recovery assertion.
No conclusion when $I_n/\log n\to1$ follows from this argument.
\end{proof}

The matching binary lower bound mentioned after the theorem has a direct
decision-theoretic justification.  Conditional on all other labels, the
rest of the graph is independent of the incident-edge row and has the
same distribution under $a$ and $b$.  Revealing the other labels and
true parameters cannot increase the Bayes risk.  A function of the
graph's spectral summary is also a function of the full graph; its risk
in that binary experiment is therefore at least
$R_{i,ab}^{\mathrm{or}}$.  This argument does not, by itself, assert a
global minimax lower bound over an unspecified parameter space.



\subsection{Full-graph reconstruction and local empirical EM}
\label{app:local-em}

\begin{proof}[Proof of Lemma~\ref{lem:full-spectral}]
The true no-self-loop mean $\E[A\mid z]$ leaves $\operatorname{col}(Z)$
invariant. In the orthonormal membership basis $ZN^{-1/2}$ its
restriction is
$N^{1/2}PN^{1/2}-\operatorname{diag}(P_{11},\ldots,P_{KK})$.
The conditioning assumption implies that its $K$ signal eigenvalues
have absolute values between $c nP_{\max}$ and $C nP_{\max}$ for large
$n$. On the orthogonal complement, the eigenvalues are $-P_{aa}$,
with absolute values at most $P_{\max}$. Let $U_*,\Lambda_*$ contain
the signal eigenpairs. Then $U_*U_*^\top=\Pi_*$ and
$\|U_*\|_{2\to\infty}=O(n^{-1/2})$.

Here $\|M\|_{2\to\infty}=\max_i\|M_i\|_2$. Fix any $H>0$.
The bounded-entry spectral norm inequality of Bandeira and van
Handel~\citep[Corollary 3.12 and Remark 3.13]{bvh2016} gives, for
$t\ge0$ and absolute constants,
\[
 \Pr\{\|A-\E[A\mid z]\|_{\mathrm{op}}
              >C\sqrt{nP_{\max}}+t\}
 \le n e^{-t^2/C}.
\]
Taking $t=C_H\sqrt{nP_{\max}}$ absorbs the factor $n$ because
$nP_{\max}\ge c\log n$. Thus the noise norm is at most
$C_H\sqrt{nP_{\max}}$ outside $O(e^{-H nP_{\max}})$.
In the notation of Abbe et al.~\citep[Theorem 2.1]{abbe2020}, the
signal separation is comparable to $nP_{\max}$, the eigenvalue
condition number is bounded, and we take
$\gamma=C_H/\sqrt{nP_{\max}}$.
Their incoherence assumption holds since
$\|\E[A\mid z]\|_{2\to\infty}=O(nP_{\max}/\sqrt n)$ and
$nP_{\max}\le n$. Row/column independence follows from the
independent upper-triangular edges.

We verify their row concentration at the same exponential confidence.
For a fixed matrix $W$ with at most $K$ columns, put
$m=\|W\|_{2\to\infty}$ and
$x=\|W\|_F/(\sqrt n m)\in(0,1]$; $m=0$ is trivial.
Every scalar column sum of $(A-\E[A\mid z])_{i*}W$ has summands bounded
by $m$ and variance at most $nP_{\max}m^2x^2$.
Bennett's inequality at threshold
$B nP_{\max}m/(1\vee\log(1/x))$ gives an exponent at least
\[
 nP_{\max}x^2
 h\!\left(\frac{B}{x^2(1\vee\log(1/x))}\right)
 \ge\frac{B}{2}nP_{\max},
 \qquad h(u)=(1+u)\log(1+u)-u,
\]
for all $0<x\le1$ and all sufficiently large fixed $B$.
For $x\le e^{-1}$ this follows from
$\log\{B/[x^2\log(1/x)]\}\ge\log(1/x)$ and
$h(u)\ge u\log(u)/2$ for sufficiently large $u$.
For $x\ge e^{-1}$ it follows by monotonicity of the Bennett exponent
in its variance bound. Choosing $B$ sufficiently large and applying
a union bound over the fixed number of columns proves their A4 with
per-row failure at most $e^{-H nP_{\max}}/n$ and
\[
 \varphi_H(x)=\frac{C_H}{1\vee\log(1/x)}\quad(x>0),
 \qquad \varphi_H(0)=0.
\]
The function is continuous and nondecreasing, and
$\varphi_H(x)/x$ is nonincreasing. Both $\gamma$ and
$\varphi_H(\gamma)$ tend to zero, so their smallness assumption holds
for large $n$. We may take both failure parameters in their theorem
to be $O(e^{-H nP_{\max}})$, after adjusting the constants.

Apply that theorem separately to the positive and negative signal
groups and concatenate the polar orthogonal alignments. Repeated
eigenvalues within either group cause no difficulty. The theorem
allows a non-low-rank mean, so the diagonal correction has been
retained. The noise norm event identifies the retained empirical
eigenpairs with these signal groups. We obtain an orthogonal $R$,
$V=UR$, and a deterministic $\nu_n\to0$ such that
\[
 \|V\|_{2\to\infty}\le C_H/\sqrt n,\qquad
 \|V-AU_*\Lambda_*^{-1}\|_{2\to\infty}\le\nu_n/\sqrt n.
\]
The same alignments satisfy Davis--Kahan's bound
$\|V-U_*\|_{\mathrm{op}}\le C_H/\sqrt{nP_{\max}}$.
Set $M=R^\top\Lambda R=V^\top AV$. Since
$\|\E[A\mid z]\|_{\mathrm{op}}=O(nP_{\max})$,
\[
 \|M-\Lambda_*\|_{\mathrm{op}}
 \le\|A-\E[A\mid z]\|_{\mathrm{op}}
 +2\|\E[A\mid z]\|_{\mathrm{op}}\|V-U_*\|_{\mathrm{op}}
 \le C_H\sqrt{nP_{\max}}.
\]
The degree union bound from Lemma~\ref{lem:sparse-expansion} also
gives $\max_i\deg(i)\le C_H nP_{\max}$ at the stated confidence.

Set $E=V-AU_*\Lambda_*^{-1}$. The exact row decomposition is
\[
 (\widetilde A-A\Pi_*)_i
 =E_iMV^\top+(AU_*)_i
       \{\Lambda_*^{-1}MV^\top-U_*^\top\}.
\]
The bracket has operator norm $O((nP_{\max})^{-1/2})$, while
$\|(AU_*)_i\|_2\le C_H nP_{\max}/\sqrt n$.
Consequently
\[
 \|\widetilde A-A\Pi_*\|_{2\to\infty}
 \le\frac{C_H}{\sqrt n}
       \{nP_{\max}\nu_n+\sqrt{nP_{\max}}\}
 =o(nP_{\max}/\sqrt n).
\]
Cauchy--Schwarz gives the uniform row $\ell_1$ bound. For a fixed
realization of the random floor $\varepsilon_n$, the clipping map
$x\mapsto\max\{x,\varepsilon_n\}$ is 1-Lipschitz. Applying this same
map to both reconstructions therefore gives the $q-q^0$ bound without
conditioning on the floor. Moreover,
\[
 |\widetilde A_{ij}|
 \le\|U_i\|_2\|\Lambda\|_{\mathrm{op}}\|U_j\|_2
 \le C_H P_{\max}.
\]
The nonnegative adjacency matrix has spectral radius
$\|\Lambda\|_{\mathrm{op}}=\lambda_1(A)$. The row sums of its mean,
together with the noise norm bound, give for large $n$
\[
 c nP_{\max}\le\|\Lambda\|_{\mathrm{op}}
 \le nP_{\max}+C_H\sqrt{nP_{\max}}
 =(1+o(1))nP_{\max}.
\]
Hence the admissible adaptive floor satisfies
$c'P_{\max}\le\varepsilon_n=c_0\max\{\|\Lambda\|_{\mathrm{op}},1\}/n
\le P_{\min}/2$ on this event for large $n$, because
$c_0<c_r/2$. This also proves the
common positive box used in the fitting arguments.

For the average approximation, let
$N_{ic}=\sum_{j:z_j=c}A_{ij}$. Since $\varepsilon_n<P_{ac}$ on this
event, clipping toward $P_{ac}$ cannot increase the coordinate error:
\[
 \|q_i^0-p_{z_i,*}\|_1
 \le\sum_c|N_{ic}-n_cP_{z_i,c}|.
\]
Let $F$ be the sum on the right over all $i$.
Its expectation is at most $Cn\sqrt{nP_{\max}}+CnP_{\max}$; the
last term accounts for the missing diagonal.
Changing one independent edge changes $F$ by at most two.
The bounded-differences inequality gives
\[
 \Pr\{F-\E F>n(nP_{\max})^{3/4}\}
 \le e^{-c(nP_{\max})^{3/2}}.
\]
This is smaller than $e^{-H nP_{\max}}$ for every fixed $H$ and large
$n$. Combining these estimates and enlarging a deterministic vanishing
$a_n$ proves every assertion on a common event of probability
$1-O(e^{-H nP_{\max}})$.
\end{proof}

\paragraph{Clipped oracle comparisons with a random floor.}
Let $G_H$ be the graph event just proved. Fix a true class $a$ and a
competitor $b$, and write
$v_c=\log(P_{ac}/P_{bc})$,
$N'_{ic}=\max\{N_{ic},n_c\varepsilon_n\}$, and
\[
 L_{i,\mathrm{clip}}^{ab}
 =\sum_cN'_{ic}v_c-\sum_c n_c(P_{ac}-P_{bc}).
\]
The floor depends on the whole graph, so its value cannot be
conditioned on while asserting independence of the block counts.
Instead, on $G_H$ it has the deterministic upper bound
$\overline\varepsilon_n=P_{\min}/2$.
Put $t_c=n_c\overline\varepsilon_n$.
For large $n$, $t_c\le\E_aN_{ic}$, also in the block containing the
excluded self-loop. For $\alpha\in[0,1]$, define the independent
count envelopes
\[
 B_c(N_{ic})=
 \begin{cases}
  e^{-\alpha v_cN_{ic}},&v_c\ge0,\\
  e^{-\alpha v_c\max\{N_{ic},t_c\}},&v_c<0.
 \end{cases}
\]
There is the pathwise inequality
\[
 e^{-\alpha L_{i,\mathrm{clip}}^{ab}}\one_{G_H}
 \le e^{\alpha\sum_c n_c(P_{ac}-P_{bc})}
       \prod_c B_c(N_{ic}).
\]
When $v_c<0$, put $\beta=-\alpha v_c\ge0$.
Jensen's inequality and $t_c\le\E_aN_{ic}$ give
\[
 \E_a e^{\beta\max\{N_{ic},t_c\}}
 \le\E_a e^{\beta N_{ic}}+e^{\beta t_c}
 \le2\E_a e^{\beta N_{ic}}.
\]
The right-hand envelopes have deterministic thresholds, so their
moments factor over the independent column-block counts.
Thus clipping costs at most $2^K$ in this restricted Chernoff moment.
A direct binomial moment expansion, including the excluded diagonal,
gives uniformly in $\alpha$
\[
 \log\E_a e^{-\alpha L_{i,\mathrm{raw}}^{ab}}
 =-D_\alpha(p_{a,-i}\Vert p_{b,-i})
      +O(nP_{\max}^2+P_{\max}),
\]
where the raw score replaces $N'_{ic}$ by $N_{ic}$.
Since $P_{\max}\to0$, this remainder is $o(nP_{\max})$.
For any deterministic $\tau_n=o(nP_{\max})$, Markov's inequality at
the Chernoff maximizer therefore yields
\begin{equation}
 \Pr_a\{L_{i,\mathrm{clip}}^{ab}\le\tau_n,\ G_H\}
 \le2^K\exp\{-D^*_{i,ab}+o(nP_{\max})\}.
 \label{eq:clipped-chernoff}
\end{equation}
This comparison handles the adaptive floor without an independence
assumption, and does not require clipping every rare-row score to
change it by $o(nP_{\max})$.

\begin{proof}[Proof of Theorem~\ref{thm:local-spectral-em}]
Choose a fixed $H>2+\limsup_n I_n/(nP_{\max})$ in
Lemma~\ref{lem:full-spectral} and work on $G_H$.
The limsup is finite by the oracle information bounds.
Constants below may depend on the model bounds and floor, but not on
the iteration. Suppose an aligned responsibility matrix $R$ has
average error at most $\gamma$, with $\gamma$ below a sufficiently
small fixed constant. Put $N_a=\sum_iR_{ia}$ and
$\bar q_a=n_a^{-1}\sum_{i:z_i=a}q_i$.
Then $|N_a-n_a|\le n\gamma$ and $N_a\ge\pi_{\min}n/2$.
The average-profile bound gives
$\|\bar q_a-p_{a*}\|_1\le Ca_n nP_{\max}$.
The M-step identity gives, deterministically,
\[
 \|\widehat p_a-\bar q_a\|_1
 \le\frac{\sum_i|R_{ia}-Z_{ia}|\|q_i\|_1}{N_a}
 +\left|\frac{n_a}{N_a}-1\right|\|\bar q_a\|_1
 \le C\gamma nP_{\max}.
\]
Consequently
\begin{equation}
 \max_a\|\widehat p_a-p_{a*}\|_1
 \le C(\gamma+a_n)nP_{\max},\qquad
 c'P_{\max}\le\widehat p_{aj}\le C_HP_{\max}.
 \label{eq:local-profile-fit}
\end{equation}
The logarithm is $C/P_{\max}$-Lipschitz on the common positive box.
Since $q_{ij}\le C_HP_{\max}$, fitted linear contrasts and offsets
therefore differ from their true-profile working counterparts by at
most $C(\gamma+a_n)nP_{\max}$.
The fitted log-weight ratios are bounded: the unconstrained weights
$N_a/n$ are bounded away from zero; a fixed positive weight floor gives
the same bound directly.

Combining the uniform $q-q^0$ bound with these parameter bounds gives,
for every node and pair,
\begin{equation}
 |\widehat\ell_i(a)-\widehat\ell_i(b)-L_{i,\mathrm{clip}}^{ab}|
 \le C(\gamma+a_n)nP_{\max}+C.
 \label{eq:local-score-fit}
\end{equation}
No independence of $R$, $q$, and the incident edges is used.
For $\gamma=\gamma_n\to0$, a wrong classification on the weak-start
and graph events implies
$L_{i,\mathrm{clip}}^{ab}\le o(nP_{\max})$ for a competitor.
Equation~\eqref{eq:clipped-chernoff} and a finite union bound prove the
one-update rate after adding the two exceptional-event probabilities.

For further updates, the conditioning assumption and the positive box
give a fixed $\kappa>0$ with
$\min_{a\ne b}D(p_{a*}\Vert p_{b*})\ge\kappa nP_{\max}$ for large
$n$. Choose a fixed basin radius $\gamma_0>0$ so that the fitted-score
error for $\gamma\le\gamma_0$ is less than
$\kappa nP_{\max}/8$ in each comparison. Choose a fixed small $t_0>0$
so that $\|q_i-p_{z_i,*}\|_1\le t_0 nP_{\max}$ costs less than
$\kappa nP_{\max}/8$ in the observation contrast.
At $q_i=p_{z_i,*}$, the population working margin is
$D(p_{z_i,*}\Vert p_{b*})\ge\kappa nP_{\max}$.
After the observation, parameter, and bounded weight corrections,
each specified row retains a correct-class margin of at least
$\kappa nP_{\max}/4$ for large $n$.
The average-profile bound leaves at most $a_n/t_0$ exceptional rows.
The next E-step thus has aligned average responsibility error at most
\[
 b_n=2a_n/t_0+2(K-1)e^{-\kappa nP_{\max}/4}\longrightarrow0.
\]
This holds uniformly throughout the fixed basin. Eventually
$b_n<\gamma_0$, so the basin is invariant. Starting from
$\gamma_n\to0$, every later responsibility matrix has error at most
$\max\{\gamma_n,b_n\}=o(1)$. Equation~\eqref{eq:local-score-fit} then
has one deterministic $o(nP_{\max})$ bound for every later score on
the same event, including a data-dependent finite stopping time.
No union bound over iterations is needed.

Finally, add the initialization failure $\delta_n$ and the graph
failure $O(e^{-H nP_{\max}})$ to the per-node Chernoff bound and average
over nodes. The chosen $H$ makes the graph failure negligible relative
to $e^{-I_n}$, and the initialization reliability preserves that
exponent. If $\liminf I_n/\log n>1$, multiplying the expected error by
$n$ and applying Markov's inequality proves exact recovery.
\end{proof}


\subsection{End-to-end likelihood initialization}
\label{app:end-to-end}

\begin{theorem}[Original growing branch for two communities]
\label{thm:two-community-growing}
Under \eqref{eq:model-conditions}--\eqref{eq:degree-condition} and the
admissible observable floor, the global-mean growing branch alone for
$K=2$, followed by at least one exact profile and weight M-step, satisfies
$\E\Mis(\widehat z,z)\le e^{-(1+o(1))I_n}$.
The same conclusion holds at any subsequent finite exact-EM stopping
time, and exact recovery holds if $\liminf I_n/\log n>1$.
\end{theorem}

\paragraph{Proof of Theorem~\ref{thm:two-community-growing}.}
Fix $H>2+\limsup_{n\to\infty}I_n/(nP_{\max})$ and work on the event of Lemma~\ref{lem:full-spectral}. All constants below may depend on $H$ and the fixed model. Enlarge its deterministic $a_n\to0$ if necessary to bound both displayed profile errors. Put $\pi_{a,n}=n_a/n$,
\[
 m_b=\sum_{a=1}^2\pi_{a,n}P_{ab},\qquad
 D_{\pi_n}(x\Vert y)=
 \sum_b\pi_{b,n}\{x_b\log(x_b/y_b)-x_b+y_b\}.
\]
The block values of $q_c^0$ are
\[
 v_{cb}=\max\left\{\frac{\sum_{j:z_j=b}A_{cj}}{n_b},
                         \varepsilon_n\right\},
 \qquad q_{cj}^0=v_{c,z_j}.
\]
This is only an analysis variable. In particular $v_c$ may be a rare extreme candidate, and we do not assume it is near its own community profile. Let
\[
 h_a(v)=\frac1{P_{\max}}\sum_b\pi_{b,n}
 \{P_{ab}\log(v_b/m_b)-(v_b-m_b)\},\qquad
 \Phi_n(v)=\sum_a\pi_{a,n}[h_a(v)]_+.
\]
The global profile mean $\overline q$ differs from $(m_{z_j})_{j=1}^n$ in $\ell_1$ norm by at most $a_n nP_{\max}$. The uniform bound on $q_c-q_c^0$ and the common positive box give
\[
 \left|[\ell_i(q_c)-\ell_i(\overline q)]-nP_{\max}h_{z_i}(v_c)\right|
 \le C\{\|q_i-p_{z_i,*}\|_1+a_n nP_{\max}\}.
\]
Here the parameter perturbation costs
$C\|q_c-q_c^0\|_1+C\|\overline q-(m_{z_j})_{j=1}^n\|_1$,
because $q_{ij}\le C P_{\max}$ and $\log$ is $C/P_{\max}$-Lipschitz. The remaining observation perturbation multiplies a uniformly bounded log ratio. Since positive part is 1-Lipschitz, summing yields the simultaneous candidate bound
\[
 \sup_c\left|\frac{\Delta_2(c)}{n^2P_{\max}}-\Phi_n(v_c)\right|\le Ca_n.
\]
Every true class has a candidate with $\|q_c-p_{a*}\|_1=o(nP_{\max})$ by the average error bound and its positive proportion. The same is true of $q_c^0$, so
$P_{\max}^{-1}\sum_b\pi_{b,n}|v_{cb}-P_{ab}|=o(1)$.
The uniform conditioning and positive community proportions imply that
\[
 g_0=\liminf_{n\to\infty}\max_a
 \frac{\pi_{a,n}}{P_{\max}}
 D_{\pi_n}\big((P_{ab})_{b=1}^2\Vert m\big)>0.
\]
Positivity follows because the conditioned block rows are uniformly separated relative to $P_{\max}$ and cannot both approach their mean at that scale. Uniform Lipschitz continuity after dividing all block values by $P_{\max}$ and the candidate bound imply that the selected candidate satisfies $\Phi_n(v_c)\ge g_0/2$ for large $n$.

The decisive two-class identity is
\[
 \sum_a\pi_{a,n}h_a(v)=-\frac1{P_{\max}}D_{\pi_n}(m\Vert v)\le0.
\]
Together with $\Phi_n(v_c)\ge g_0/2$, it forces exactly one of $h_1(v_c),h_2(v_c)$ to be positive. Moreover
\[
 h_{\rm pos}(v_c)\ge g_0/2,\qquad
 h_{\rm neg}(v_c)\le-g_0/2,
\]
since each weighted positive or negative magnitude is at least $g_0/2$ and each proportion is at most one. The initial two mixture weights are equal, so they do not alter this score comparison.

Take $\tau_n=\sqrt{a_n}+(nP_{\max})^{-1}$. At most $(a_n/\tau_n)n=o(n)$ observations have $\|q_i-p_{z_i,*}\|_1>\tau_n nP_{\max}$. Every other observation has the correct initial component margin at least $g_0 nP_{\max}/4$ after the above score perturbations. After a common label alignment, the first E-step therefore satisfies
\[
 \frac1n\sum_{i,a}|R_{ia}-Z_{ia}|
 \le \frac{2a_n}{\tau_n}+2e^{-g_0 nP_{\max}/4}=o(1).
\]
The graph event has failure $O(e^{-H nP_{\max}})$, so these responsibilities meet the reliability condition in Theorem~\ref{thm:local-spectral-em}. The evaluated fitting loop always performs its first M-step before it may stop. That theorem gives the claimed exponent for the first resulting decode and all subsequent finite updates, including the evaluated stopping rule. Its exact-recovery implication also applies. \qed

\subsection{Frozen-candidate greedy gains}

\begin{proposition}[Greedy gain guarantee]
\label{prop:greedy-gain}
Let $s_{ic}$ be fixed finite candidate scores and let $S_0$ be a fixed
nonempty seed set.  For candidate sets $T$, define
\[
 G(T)=\sum_i\max_{c\in S_0\cup T}s_{ic}
      -\sum_i\max_{c\in S_0}s_{ic}.
\]
Then $G$ is normalized, monotone, and submodular.  If $m\ge1$ greedy
steps select a set $T_m$ and $T^*$ maximizes $G$ over sets of at most
$m$ candidates, then
\[
 G(T_m)\ge\{1-(1-1/m)^m\}G(T^*)\ge(1-e^{-1})G(T^*).
\]
\end{proposition}

\begin{proof}
Normalization and monotonicity follow immediately from the definition.
The marginal gain of candidate $c$ is
\[
 G(T\cup\{c\})-G(T)
 =\sum_i\left(s_{ic}-\max_{b\in S_0\cup T}s_{ib}\right)_+.
\]
If $T\subseteq T'$, each maximum in the right side increases, so the
marginal gain decreases.  This proves submodularity.
For the greedy set $T_t$ at time $t$, monotonicity and telescoping
diminishing returns give
\begin{align*}
 G(T^*)-G(T_t)
 &\le G(T_t\cup T^*)-G(T_t)\\
 &\le\sum_{c\in T^*\setminus T_t}
        \{G(T_t\cup\{c\})-G(T_t)\}\\
 &\le m\{G(T_{t+1})-G(T_t)\}.
\end{align*}
Starting from $G(T_0)=0$ and iterating this inequality proves the
claimed factor.  The final inequality follows from
$(1-1/m)^m\le e^{-1}$, including the case $m=1$.
\end{proof}

This is the standard cardinality-constrained submodular greedy
guarantee~\cite{nemhauser1978}.  It controls added explanation relative
to the fixed initial seed, not the multiplicative approximation of a
possibly negative total log score.  It does not apply unchanged when
intermediate EM steps alter the candidate scores.

\subsection{Noiseless profile selection}

\begin{proposition}[Population profile selection]
\label{prop:noiseless-seeding}
Suppose $q_i=\theta_{z_i}$, where the $K$ positive profile vectors
$\theta_1,\ldots,\theta_K$ are pairwise distinct and every class is
nonempty.  The candidate pool consists of these node profiles.  From
any first seed belonging to this candidate pool, $K-1$ subsequent global-gain greedy steps select all
$K$ different profiles.  Maximum working-score classification using
those profiles recovers $z$ up to permutation.
\end{proposition}

\begin{proof}
For $x,\widehat p>0$ coordinatewise, write
$\ell_x(\widehat p)=\sum_jx_j\log\widehat p_j-\sum_j\widehat p_j$.  Then
\[
 \ell_x(x)-\ell_x(\widehat p)
 =\sum_j\{x_j\log(x_j/\widehat p_j)-x_j+\widehat p_j\}
 =D(x\Vert\widehat p)\ge0,
\]
with equality precisely when $x=\widehat p$; the inequality follows from
$\log u\le u-1$.  If a profile has already been selected, selecting a
duplicate has zero gain.  If profile $\theta_b$ has not been selected,
then each node in class $b$ gains
$\min_{a\in S}D(\theta_b\Vert\theta_a)>0$
when it is added.  Therefore some unselected profile has strictly
positive gain, and greedy cannot select a duplicate before all distinct
profiles have been selected.  The same strict divergence inequality
makes each node's own profile its unique best score after selection.
\end{proof}

The proposition concerns the frozen node-profile selection control in an abstract noiseless profile model. It does not directly cover Algorithm~1, whose initial mean need not be a node profile and whose intermediate EM fits change the parameters.  It does
not replace a stability analysis for random spectral profiles, and it
does not erase the diagonal distinction in a no-self-loop SBM.

\subsection{Exact working EM and accepted replacements}

\begin{proposition}[Objective convergence of exact working EM]
\label{prop:em-monotonicity}
For fixed strictly positive profiles $q_1,\ldots,q_n$, let
\[
 \mathcal L(\widehat\pi,\widehat p)=\sum_i\log\sum_{a=1}^K
       \widehat\pi_a\exp\{\ell_i(\widehat p_a)\},\qquad
 \ell_i(\widehat p_a)=\sum_jq_{ij}\log\widehat p_{aj}-\sum_j\widehat p_{aj}.
\]
Starting with positive parameters and positive mixture weights, the
exact E-step and unconstrained exact M-step
\begin{align*}
 r_{ia}&=\frac{\widehat\pi_a e^{\ell_i(\widehat p_a)}}
                   {\sum_b\widehat\pi_b e^{\ell_i(\widehat p_b)}},\qquad
 N_a=\sum_i r_{ia},\\
 \widehat\pi_a^{\mathrm{new}}&=N_a/n,\qquad
 \widehat p_{aj}^{\mathrm{new}}=\frac{\sum_i r_{ia}q_{ij}}{N_a}
\end{align*}
make $\mathcal L$ nondecreasing.  Its numerical values converge to a
finite limit.  The same monotonicity statement holds for an exact
M-step maximizing over any stated constraint set containing the current
parameters, and for replacement proposals accepted only when their
final $\mathcal L$ improves.
\end{proposition}

\begin{proof}
For row-wise probability vectors $r_i$, define the variational objective
\[
 \mathcal F(r,\widehat\pi,\widehat p)=
 \sum_{i,a}r_{ia}\{\log\widehat\pi_a+\ell_i(\widehat p_a)-\log r_{ia}\}.
\]
The log-sum inequality gives
$\mathcal F(r,\widehat\pi,\widehat p)\le\mathcal L(\widehat\pi,\widehat p)$, with equality
for the E-step responsibilities.  For fixed $r$, maximizing
$\sum_aN_a\log\widehat\pi_a$ over the simplex gives $\widehat\pi_a=N_a/n$.
For each profile coordinate, maximizing
$(\sum_i r_{ia}q_{ij})\log\widehat p_{aj}-N_a\widehat p_{aj}$ gives the displayed
weighted mean.  These are interior maxima because all inputs and
responsibilities are positive.  Thus the E-step makes the variational
bound tight and the M-step raises it, proving monotonicity.

The saturated-score inequality used in the preceding proof implies
$\ell_i(\widehat p_a)\le\ell_i(q_i)$ for every $i,a$.  Since
$\sum_a\widehat\pi_a=1$,
\[
 \log\sum_a\widehat\pi_a e^{\ell_i(\widehat p_a)}\le\ell_i(q_i).
\]
Consequently $\mathcal L\le\sum_i\ell_i(q_i)<\infty$.
A nondecreasing real sequence bounded above has a finite limit.
For a constrained exact M-step the current parameters remain feasible,
so the same variational argument applies.  The replacement assertion
follows directly from its acceptance rule.
\end{proof}

This proposition proves convergence of objective values.  It proves
neither convergence to the global maximum nor recovery of the true
labels. The implementation uses an exact active-set maximizer on the
floored simplex, so its weight constraint is covered by the constrained
version of this proposition. Its finite stopping tolerance and
floating-point accuracy are assessed separately in Section~\ref{sec:experiments}.

\begin{proposition}[Profile hull and spectral dimension]
\label{prop:profile-hull}
Every responsibility-weighted profile M-step produces a profile in the convex hull of the rows of $q$. If the unmodified reconstruction $q=U\Lambda U^\top$ is used and the resulting means are positive, all fitted profiles lie in its row space, of dimension at most $K$. Entrywise positive clipping can increase this dimension.
\end{proposition}
\begin{proof}
The coefficients $R_{ia}/N_a$ are nonnegative and sum to one, so the update is a row convex combination. Every such combination also lies in the row space. The unmodified reconstruction has rank at most $K$. The clipping operation is nonlinear and has no corresponding rank-preservation identity.
\end{proof}
